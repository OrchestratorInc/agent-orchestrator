func TestServeEvidence(t *testing.T) {
	addr := os.Getenv("EVIDENCE_ADDR")
	if addr == "" {
		t.Skip("EVIDENCE_ADDR unset")
	}
	created, _ := time.Parse(time.RFC3339, os.Getenv("REAL_CREATED"))
	userHome, _ := os.UserHomeDir()
	dataDir := filepath.Join(userHome, ".ao", "data")
	plugin := claudecodeagent.New()
	claude := &agentaccount.Claude{
		Binary: plugin.ResolveBinary, DeviceHome: filepath.Join(userHome, ".claude"), DeviceConfig: filepath.Join(userHome, ".claude.json"),
		DeviceEnv: func() map[string]string { return plugin.RuntimeCredentialEnv(dataDir) },
	}
	id := os.Getenv("REAL_ID")
	st := &store{
		accounts: []domain.AgentAccount{{ID: id, Provider: domain.AgentAccountProviderClaude, Home: os.Getenv("REAL_HOME"), Email: os.Getenv("REAL_EMAIL"), CreatedAt: created}},
		def:      map[domain.AgentAccountProvider]string{domain.AgentAccountProviderClaude: id},
	}
	svc := agentaccounts.New(st, []ports.AgentAccountProviderCLI{claude}, filepath.Join(dataDir, "agent-accounts"), nil, nil)
	ctrl := &controllers.AgentAccountsController{Svc: svc}

	live, _ := url.Parse(os.Getenv("LIVE_DAEMON"))
	proxy := httputil.NewSingleHostReverseProxy(live)
	direct := proxy.Director
	proxy.Director = func(req *http.Request) {
		direct(req)
		req.Header.Del("Origin")
		req.Header.Del("Referer")
	}
	r := chi.NewRouter()
	r.Use(func(next http.Handler) http.Handler {
		return http.HandlerFunc(func(w http.ResponseWriter, req *http.Request) {
			w.Header().Set("Access-Control-Allow-Origin", "*")
			w.Header().Set("Access-Control-Allow-Headers", "*")
			if req.Method == http.MethodOptions {
				w.WriteHeader(http.StatusNoContent)
				return
			}
			if req.Method != http.MethodGet {
				http.Error(w, "evidence harness is read-only", http.StatusMethodNotAllowed)
				return
			}
			next.ServeHTTP(w, req)
		})
	})
	r.Get("/api/v1/agents/accounts", func(w http.ResponseWriter, req *http.Request) {
		sub := chi.NewRouter()
		sub.Route("/api/v1", ctrl.Register)
		sub.ServeHTTP(w, req)
	})
	r.NotFound(func(w http.ResponseWriter, req *http.Request) {
		proxy.ServeHTTP(w, req)
	})
	srv := &http.Server{Addr: addr, Handler: r}
	t.Logf("serving on %s", addr)
	if err := srv.ListenAndServe(); err != nil {
		t.Fatal(err)
	}
}
